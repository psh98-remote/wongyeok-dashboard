import { createClient } from 'https://esm.sh/@supabase/supabase-js@2' Deno.serve(async (req) => { try { const supabaseUrl = Deno.env.get('SUPABASE_URL')! const supabaseKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')! const sb = createClient(supabaseUrl, supabaseKey, { auth: { persistSession: false } }) const now = new Date() const year = String(now.getFullYear()) const month = String(now.getMonth() + 1).padStart(2, '0') const monthNum = now.getMonth() + 1 // systems 기준으로 점검 생성 const { data: systems, error: sysErr } = await sb .from('systems') .select('*') if (sysErr) throw new Error('systems 조회 실패: ' + sysErr.message) const toInsert = [] for (const s of systems) { if (!s.cycle || s.cycle === '장애시(필요시)') continue if (s.cycle === '분기') { const baseMonth = s.base_month || '' if (baseMonth) { const validMonths = baseMonth .split(/[,\/\s]+/) .map((m: string) => parseInt(m.trim())) .filter((m: number) => !isNaN(m)) if (!validMonths.includes(monthNum)) continue } else { if (![3, 6, 9, 12].includes(monthNum)) continue } } // 이미 해당 월/시스템 데이터 있는지 확인 const { data: existing } = await sb .from('inspections') .select('id') .eq('system_id', s.id) .eq('year', year) .eq('month', month) if (existing && existing.length > 0) continue toInsert.push({ system_id: s.id, customer_name: s.customer_name, system_name: s.system_name, year, month, product: s.product || '', version: s.version || '', manager: s.manager || '', status: '미완료', inspection_date: '', note: s.specific_day || '', }) } if (toInsert.length > 0) { const { error: insErr } = await sb.from('inspections').insert(toInsert) if (insErr) throw new Error('insert 실패: ' + insErr.message) } return new Response( JSON.stringify({ success: true, created: toInsert.length, year, month, systems: toInsert.map(i => `${i.customer_name} - ${i.system_name}`) }), { headers: { 'Content-Type': 'application/json' } } ) } catch (e) { return new Response( JSON.stringify({ success: false, error: e.message }), { status: 500, headers: { 'Content-Type': 'application/json' } } ) } })
